_check_rate_limit_keyed(store, key, max_requests=60, window_seconds=60):
    now = time.time()
    entry = store.get(key, [])
    entry = [t for t in entry if now - t < window_seconds]
    if len(entry) >= max_requests:
        store[key] = entry
        return False
    entry.append(now)
    store[key] = entry
    if len(store) > 5000:
        _cutoff = now - window_seconds
        for _k in list(store.keys()):
            if not store[_k] or max(store[_k]) < _cutoff:
                del store[_k]
    return True
